--- DASHBOARD 1: REVENUE PERFORMANCE DASHBOARD ---

* Total Revenue = SUM(vw_recharge_clean[amount])
* ARPU = DIVIDE([Total Revenue],DISTINCTCOUNT(vw_recharge_clean[customer_id]))
* Previous Month Revenue = CALCULATE([Total Revenue],DATEADD(vw_date_clean[date],-1,MONTH))
* MoM Growth % = DIVIDE([Total Revenue]-[Previous Month Revenue],[Previous Month Revenue])
* previously active customers = CALCULATE(DISTINCTCOUNT(vw_customer_clean[Customer_id]),FILTER(vw_customer_clean,CALCULATE(COUNTROWS(vw_recharge_clean),vw_recharge_clean[recharge_date]>0) ||CALCULATE(COUNTROWS(vw_usage_clean),vw_usage_clean[data_mb]>0 || vw_usage_clean[call_minutes]>0)))
* Active Customers = [previously active customers]
* Customer Revenue = SUM(vw_recharge_clean[amount])

--- DASHBOARD 2: CUSTOMER CHURN ANALYSIS DASHBOARD ---

* Total Customers = DISTINCTCOUNT(vw_customer_clean[Customer_id])
* Churn Customers = COUNTROWS(vw_churn_customers)
* churn % = DIVIDE([Churn Customers],[previously active customers])
* Revenue Loss = CALCULATE([Total Revenue],vw_churn_customers)
* Monthly Churn Count = DISTINCTCOUNT(vw_churn_customers[customer_id])
* Customer Revenue = SUM(vw_recharge_clean[amount])
* Recharge Count = COUNTROWS(vw_recharge_clean)

--- CALCULATED COLUMNS (DATA MODELING) ---

*  Churn Date = MAX(vw_churn_customers[last_recharge_date].[Date],vw_churn_customers[last_active_day].[Date])
*  Customer Type = SWITCH(TRUE(),[Recharge Count]=1,"One-time",[Recharge Count]<=5,"Low Repeat","High Repeat")
